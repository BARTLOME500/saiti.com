function freq(i){const k=i%15;return 196*Math.pow(2,(PENT[k%5]+12*Math.floor(k/5))/12)}
function tone(i){
  try{
    ctx=ctx||new (window.AudioContext||window.webkitAudioContext)();
    if(ctx.state==="suspended")ctx.resume();
    const t=ctx.currentTime,o=ctx.createOscillator(),g=ctx.createGain();
    o.type="triangle";o.frequency.value=freq(i);
    g.gain.setValueAtTime(0,t);g.gain.linearRampToValueAtTime(.35,t+.02);g.gain.exponentialRampToValueAtTime(.001,t+.9);

}

  }
